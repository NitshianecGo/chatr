// ====================================================================== // НАСТРОЙКА СОЕДИНЕНИЯ И ОЧЕРЕДИ // ====================================================================== function setupConnectionHandlers() { // Отключаем кнопку отправки до тех пор, пока P2P-канал не станет active sendChatBtn.disabled = true; chatInput.disabled = true; chatInput.placeholder = "Установка соединения..."; conn.on('open', () => { console.log('✅ DataChannel открыт!'); statusDisplay.textContent = '✅ Соединение для чата установлено!'; // Разблокируем ввод sendChatBtn.disabled = false; chatInput.disabled = false; chatInput.placeholder = "Введите сообщение..."; // Обмениваемся публичными ключами шифрования conn.send({ type: 'key', publicKey: myKeyPair.publicKey }); }); conn.on('data', handleIncomingData); conn.on('close', () => { conn = null; sendChatBtn.disabled = true; chatInput.disabled = true; chatInput.placeholder = "Соединение закрыто"; statusDisplay.textContent = '⚠️ Соединение чата разорвано.'; }); conn.on('error', (err) => { console.error('DataConnection error:', err); statusDisplay.textContent = '❌ Ошибка соединения: ' + err.message; }); } // ====================================================================== // ИСПРАВЛЕННАЯ ФУНКЦИЯ ОТПРАВКИ // ====================================================================== function sendMessage(text, mediaType = 'text', fileData = null) { // Проверяем наличие объекта и реальный статус WebRTC DataChannel if (!conn || !conn.open) { statusDisplay.textContent = '⚠️ Соединение ещё устанавливается, подождите пару секунд...'; console.warn('Попытка отправки до открытия conn.open'); return; } const rawContent = fileData || text; if (!rawContent && mediaType === 'text') return; let encrypted = null; let finalContent = rawContent; if (peerPublicKey) { encrypted = encryptMessage(rawContent, peerPublicKey, myKeyPair.secretKey); if (encrypted) finalContent = '[Encrypted]'; } const msg = { type: 'message', sender: 'Я', content: finalContent, mediaType, timestamp: Date.now(), encrypted }; try { conn.send(msg); addMessage('Я', rawContent, mediaType, msg.timestamp); } catch (e) { console.error('Send error:', e); statusDisplay.textContent = '❌ Ошибка отправки: ' + e.message; } }
